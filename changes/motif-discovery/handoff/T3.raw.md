# FILE: src/motif.py
"""K-mer presence motif discovery with cluster-level coverage and dinucleotide-shuffle nulls."""
from __future__ import annotations

import logging
import math
import os
import sys
import time
from multiprocessing import Pool

import numpy as np
import pandas as pd

logger = logging.getLogger(__name__)

_LUT = np.full(256, 255, dtype=np.uint8)
for _i, _ch in enumerate("ACGT"):
    _LUT[ord(_ch)] = _i
    _LUT[ord(_ch.lower())] = _i
_LUT[ord("U")] = 3
_LUT[ord("u")] = 3

REGION_LEN = 300
_RESULT_COLUMNS = [
    "k",
    "region",
    "kmer",
    "obs_clusters",
    "coverage_frac",
    "n_families",
    "null_mean",
    "log2_enrichment",
    "p_emp",
    "p_pooled",
    "q",
    "rank",
]


def _configure_progress_logging() -> None:
    """Attach a stderr handler to this module's logger once (progress lines)."""
    if not logger.handlers:
        handler = logging.StreamHandler(sys.stderr)
        handler.setFormatter(logging.Formatter("%(asctime)s %(message)s"))
        logger.addHandler(handler)
        logger.setLevel(logging.INFO)
        logger.propagate = False


def _log_progress(label: str, done: int, total: int, t0: float) -> None:
    """Log 'label done/total (pct%) elapsed=Xs' (elapsed in seconds)."""
    logger.info(
        "%s %d/%d (%.0f%%) elapsed=%.1fs", label, done, total, 100.0 * done / total,
        time.time() - t0,
    )


def dinuc_shuffle(seq: str, rng: np.random.Generator) -> str:
    """Altschul-Erickson dinucleotide-preserving shuffle of a string; returns a string of equal length."""
    n = len(seq)
    if n < 3:
        return seq
    letters = sorted(set(seq))
    edges: dict[str, list[str]] = {c: [] for c in letters}
    for a, b in zip(seq, seq[1:]):
        edges[a].append(b)
    last = seq[-1]
    roots = [c for c in letters if c != last and edges[c]]
    # Choose a last-exit edge per vertex; accept only if these edges form a spanning
    # arborescence directed to `last` (guarantees a valid Eulerian path).
    while True:
        last_edge = {c: edges[c][int(rng.integers(len(edges[c])))] for c in roots}
        ok = True
        for c in roots:
            cur = c
            for _ in range(len(letters)):
                if cur == last:
                    break
                cur = last_edge[cur]
            if cur != last:
                ok = False
                break
        if ok:
            break
    order: dict[str, list[str]] = {}
    for c in letters:
        lst = list(edges[c])
        if c in last_edge:
            lst.remove(last_edge[c])
        perm = rng.permutation(len(lst))
        lst = [lst[i] for i in perm]
        if c in last_edge:
            lst.append(last_edge[c])
        order[c] = lst
    pos = {c: 0 for c in letters}
    cur = seq[0]
    out = [cur]
    for _ in range(n - 1):
        nxt = order[cur][pos[cur]]
        pos[cur] += 1
        out.append(nxt)
        cur = nxt
    return "".join(out)


def _encode_seq(seq: str, k: int) -> np.ndarray:
    """Sorted unique base-4 codes (A0 C1 G2 T/U3) of ACGT-only k-mers in one sequence."""
    arr = _LUT[np.frombuffer(seq.encode("ascii", errors="replace"), dtype=np.uint8)]
    n = arr.size - k + 1
    if n <= 0:
        return np.empty(0, dtype=np.int64)
    bad = arr == 255
    codes = np.zeros(n, dtype=np.int64)
    badw = np.zeros(n, dtype=bool)
    for j in range(k):
        codes = codes * 4 + arr[j : j + n].astype(np.int64)
        badw |= bad[j : j + n]
    return np.unique(codes[~badw])


def kmer_presence(seqs: list[str], k: int) -> list[np.ndarray]:
    """Per sequence, sorted unique int64 base-4 codes of k-mers (ACGT; U read as T; N-containing skipped)."""
    if k < 1:
        raise ValueError("k must be >= 1")
    return [_encode_seq(s, k) for s in seqs]


def _count_distinct(presence: list[np.ndarray], groups: np.ndarray, k: int) -> np.ndarray:
    """For each k-mer code, number of distinct groups with >=1 sequence containing it."""
    groups = np.asarray(groups)
    if len(presence) != groups.size:
        raise ValueError("presence and group labels must have equal length")
    size = 4**k
    if not presence:
        return np.zeros(size, dtype=np.int64)
    _, gi = np.unique(groups, return_inverse=True)
    lens = np.fromiter((p.size for p in presence), dtype=np.int64, count=len(presence))
    if lens.sum() == 0:
        return np.zeros(size, dtype=np.int64)
    codes = np.concatenate(presence)
    pair = np.repeat(gi.astype(np.int64), lens) * size + codes
    uniq = np.unique(pair)
    return np.bincount(uniq % size, minlength=size).astype(np.int64)


def cluster_coverage(presence: list[np.ndarray], cluster_ids: np.ndarray, k: int) -> np.ndarray:
    """Number of distinct clusters containing each k-mer (length 4**k int array; a cluster counts once)."""
    return _count_distinct(presence, cluster_ids, k)


def family_breadth(presence: list[np.ndarray], family_labels: np.ndarray, k: int) -> np.ndarray:
    """Number of distinct families with >=1 sequence containing each k-mer (length 4**k int array)."""
    return _count_distinct(presence, family_labels, k)


def _trim(seq: str, region: str | None) -> str:
    """Restrict a sequence to None (whole), '5p300' (first 300 nt) or '3p300' (last 300 nt)."""
    if region is None:
        return seq
    if region == "5p300":
        return seq[:REGION_LEN]
    if region == "3p300":
        return seq[-REGION_LEN:]
    raise ValueError(f"unknown region: {region!r}")


def _null_worker(args: tuple) -> dict[int, np.ndarray]:
    """One null replicate: shuffle all (already trimmed) seqs, return {k: cluster coverage}."""
    seqs, cluster_ids, ks, ss = args
    rng = np.random.default_rng(ss)
    shuffled = [dinuc_shuffle(s, rng) for s in seqs]
    return {k: cluster_coverage(kmer_presence(shuffled, k), cluster_ids, k) for k in ks}


def null_coverage_multi(
    seqs: list[str],
    cluster_ids: np.ndarray,
    ks: tuple[int, ...],
    n_null: int = 100,
    seed: int = 42,
    n_jobs: int | None = None,
    region: str | None = None,
    label: str = "null",
) -> dict[int, np.ndarray]:
    """Null cluster coverage for several k from the same shuffles; {k: (n_null, 4**k) int32}."""
    trimmed = [_trim(s, region) for s in seqs]  # trim BEFORE shuffling
    cids = np.asarray(cluster_ids)
    seeds = np.random.SeedSequence(seed).spawn(n_null)
    tasks = [(trimmed, cids, tuple(ks), ss) for ss in seeds]
    n_jobs = os.cpu_count() or 1 if n_jobs is None else n_jobs
    n_jobs = max(1, min(n_jobs, n_null))
    out = {k: np.zeros((n_null, 4**k), dtype=np.int32) for k in ks}
    t0 = time.time()
    step = max(1, n_null // 10)

    def _store(i: int, res: dict[int, np.ndarray]) -> None:
        for k in ks:
            out[k][i] = res[k]
        if (i + 1) % step == 0 or i + 1 == n_null:
            _log_progress(label, i + 1, n_null, t0)

    if n_jobs == 1:
        for i, t in enumerate(tasks):
            _store(i, _null_worker(t))
    else:
        with Pool(n_jobs) as pool:
            for i, res in enumerate(pool.imap(_null_worker, tasks, chunksize=1)):
                _store(i, res)
    return out


def null_coverage(
    seqs: list[str],
    cluster_ids: np.ndarray,
    k: int,
    n_null: int = 100,
    seed: int = 42,
    n_jobs: int | None = None,
    region: str | None = None,
) -> np.ndarray:
    """Cluster coverage of dinuc-shuffled seqs; (n_null, 4**k) int32; region trimmed before shuffling."""
    return null_coverage_multi(seqs, cluster_ids, (k,), n_null, seed, n_jobs, region)[k]


def bh_fdr(p: np.ndarray) -> np.ndarray:
    """Benjamini-Hochberg q-values for p-values in [0,1]; NaNs are ignored and returned as NaN."""
    p = np.asarray(p, dtype=float)
    q = np.full(p.shape, np.nan)
    ok = ~np.isnan(p)
    pv = p[ok]
    m = pv.size
    if m == 0:
        return q
    order = np.argsort(pv, kind="stable")
    adj = pv[order] * m / np.arange(1, m + 1)
    adj = np.minimum.accumulate(adj[::-1])[::-1]
    res = np.empty(m)
    res[order] = np.clip(adj, 0.0, 1.0)
    q[ok] = res
    return q


def _decode_kmers(size: int, k: int) -> list[str]:
    """Decode all codes 0..size-1 to ACGT strings of length k."""
    codes = np.arange(size, dtype=np.int64)
    powers = 4 ** np.arange(k - 1, -1, -1, dtype=np.int64)
    digits = (codes[:, None] // powers) % 4
    letters = np.array(list("ACGT"))[digits]
    return ["".join(row) for row in letters]


def _pooled_p(obs: np.ndarray, null: np.ndarray, null_mean: np.ndarray) -> np.ndarray:
    """Pooled-null p: k-mers sharing round(null_mean) share one pooled null distribution.

    p = (1 + #pooled>=obs) / (1 + pooled.size), pooled = all null values of all k-mers in the bin.
    Assumes k-mers with similar null mean are exchangeable (variance heterogeneity ignored).
    """
    bins = np.rint(null_mean).astype(np.int64)
    p = np.empty(obs.size)
    for b in np.unique(bins):
        idx = np.flatnonzero(bins == b)
        pooled = np.sort(null[:, idx].ravel())
        n_ge = pooled.size - np.searchsorted(pooled, obs[idx], side="left")
        p[idx] = (1.0 + n_ge) / (1.0 + pooled.size)
    return p


def empirical_test(obs: np.ndarray, null: np.ndarray, min_clusters: int = 3) -> pd.DataFrame:
    """Per-k-mer enrichment of observed cluster counts vs null (n_null, 4**k); BH q on eligible only."""
    obs = np.asarray(obs, dtype=float)
    null = np.asarray(null, dtype=float)
    size = obs.size
    k = int(round(math.log(size, 4))) if size > 1 else 0
    if k < 1 or 4**k != size or null.ndim != 2 or null.shape[1] != size:
        raise ValueError("obs must have length 4**k and null shape (n_null, 4**k)")
    n_null = null.shape[0]
    null_mean = null.mean(axis=0)
    null_sd = null.std(axis=0, ddof=1) if n_null > 1 else np.zeros(size)
    p_emp = (1.0 + (null >= obs).sum(axis=0)) / (1.0 + n_null)
    with np.errstate(divide="ignore", invalid="ignore"):
        z = (obs - null_mean) / null_sd
    p_z = np.array([np.nan if not np.isfinite(v) else 0.5 * math.erfc(v / math.sqrt(2)) for v in z])
    p_pooled = _pooled_p(obs, null, null_mean)
    eligible = obs >= min_clusters
    q = np.full(size, np.nan)
    if eligible.any():
        q[eligible] = bh_fdr(p_pooled[eligible])
    return pd.DataFrame(
        {
            "kmer": _decode_kmers(size, k),
            "obs": obs.astype(np.int64),
            "null_mean": null_mean,
            "null_sd": null_sd,
            "log2_enrichment": np.log2((obs + 0.5) / (null_mean + 0.5)),
            "p_emp": p_emp,
            "p_z": p_z,
            "p_pooled": p_pooled,
            "q": q,
            "eligible": eligible,
        }
    )


def run_discovery(
    seqs: list[str],
    cluster_ids: np.ndarray,
    family_labels: np.ndarray,
    ks: tuple[int, ...] = (4, 5, 6, 7, 8),
    regions: tuple[str | None, ...] = (None, "5p300", "3p300"),
    n_null: int = 100,
    seed: int = 42,
    n_jobs: int | None = None,
    min_clusters: int = 3,
) -> pd.DataFrame:
    """Discover enriched k-mers per (k, region); returns eligible k-mers ranked within each (k, region)."""
    cluster_ids = np.asarray(cluster_ids)
    family_labels = np.asarray(family_labels)
    if not (len(seqs) == cluster_ids.size == family_labels.size):
        raise ValueError("seqs, cluster_ids, family_labels must have equal length")
    _configure_progress_logging()
    n_clusters = max(1, np.unique(cluster_ids).size)
    t0 = time.time()
    frames: list[pd.DataFrame] = []
    total = len(regions) * len(ks)
    done = 0
    for region in regions:
        trimmed = [_trim(s, region) for s in seqs]
        null = null_coverage_multi(
            seqs, cluster_ids, tuple(ks), n_null, seed, n_jobs, region, label=f"null[{region}]"
        )
        for k in ks:
            pres = kmer_presence(trimmed, k)
            obs = cluster_coverage(pres, cluster_ids, k)
            fam = family_breadth(pres, family_labels, k)
            res = empirical_test(obs, null[k], min_clusters)
            res = res[res["eligible"]].copy()
            res["n_families"] = fam[res.index.to_numpy()]
            res["k"] = k
            res["region"] = "all" if region is None else region
            res["obs_clusters"] = res["obs"]
            res["coverage_frac"] = res["obs"] / n_clusters
            frames.append(res)
            done += 1
            _log_progress("tested (k,region)", done, total, t0)
    if not frames:
        return pd.DataFrame(columns=_RESULT_COLUMNS)
    out = pd.concat(frames, ignore_index=True)
    out = out.sort_values(
        ["k", "region", "q", "log2_enrichment", "n_families"],
        ascending=[True, True, True, False, False],
        kind="stable",
    )
    out["rank"] = out.groupby(["k", "region"]).cumcount() + 1
    return out[_RESULT_COLUMNS].reset_index(drop=True)

from __future__ import annotations

import sys

from pathlib import Path

import motif  # noqa: E402


def _markov_seq(rng: np.random.Generator, length: int, cum: np.ndarray) -> str:
    """Random sequence from a first-order Markov chain with cumulative transition matrix cum."""
    u = rng.random(length)
    state = int(rng.integers(4))
    out = []
    for i in range(length):
        out.append("ACGT"[state])
        state = min(int(np.searchsorted(cum[state], u[i])), 3)
    return "".join(out)


def _uniform_seq(rng: np.random.Generator, length: int) -> str:
    """Uniform iid ACGT sequence."""
    return "".join(rng.choice(list("ACGT"), size=length))


@pytest.fixture
def tiny() -> dict:
    """Tiny synthetic dataset: 4 sequences, 2 clusters, 2 families."""
    return {
        "seqs": ["ACGTACGTAC", "ACGTACGTAC", "TTTTGGGGCC", "ACGNACGTAA"],
        "clusters": np.array([0, 0, 1, 1]),

    }


def test_shuffle_preserves_dinucleotides_and_first_letter() -> None:
    rng = np.random.default_rng(0)
    base = np.random.default_rng(1)
    changed = False
    for _ in range(30):
        s = _uniform_seq(base, int(base.integers(5, 200)))
        t = motif.dinuc_shuffle(s, rng)
        assert len(t) == len(s)
        assert t[0] == s[0] and t[-1] == s[-1]
        assert Counter(zip(s, s[1:])) == Counter(zip(t, t[1:]))
        changed |= t != s
    assert changed
    assert motif.dinuc_shuffle("", rng) == ""
    assert motif.dinuc_shuffle("A", rng) == "A"
    assert motif.dinuc_shuffle("AAAA", rng) == "AAAA"


def test_bh_fdr_known_values() -> None:
    q = motif.bh_fdr(np.array([0.01, 0.04, 0.03, 0.005]))
    np.testing.assert_allclose(q, [0.02, 0.04, 0.04, 0.02])
    q = motif.bh_fdr(np.array([0.04, 0.01, 0.9]))
    np.testing.assert_allclose(q, [0.06, 0.03, 0.9])
    q = motif.bh_fdr(np.array([0.5, np.nan]))
    assert q[0] == 0.5 and np.isnan(q[1])
    assert motif.bh_fdr(np.array([])).size == 0

def test_identical_sequences_in_one_cluster_count_once() -> None:

    pres = motif.kmer_presence(seqs, 4)
    cov = motif.cluster_coverage(pres, np.zeros(10, dtype=int), 4)
    assert cov.shape == (256,) and cov.max() == 1
    assert cov.sum() == pres[0].size
    cov2 = motif.cluster_coverage(pres, np.array([0] * 5 + [1] * 5), 4)

    fam = motif.family_breadth(pres, np.array(["x"] * 10), 4)

def test_planted_motif_recovered_as_top_hit() -> None:
    rng = np.random.default_rng(7)
    trans = rng.dirichlet(np.ones(4) * 2.0, size=4)
    cum = np.cumsum(trans, axis=1)
    motif_seq = "GATTACA"
    n_clusters, per, length = 100, 2, 300  # reduced from 30x100 seqs for runtime
    planted = set(rng.choice(n_clusters, size=30, replace=False).tolist())
    seqs, cids = [], []
    for c in range(n_clusters):
        for j in range(per):
            s = _markov_seq(rng, length, cum)
            if c in planted and j == 0:
                pos = int(rng.integers(0, length - 7))
                s = s[:pos] + motif_seq + s[pos + 7 :]
            seqs.append(s)
            cids.append(c)
    fams = np.array([f"F{c % 8}" for c in cids])
    res = motif.run_discovery(
        seqs, np.array(cids), fams, ks=(7,), regions=(None,), n_null=100, seed=42, n_jobs=2
    )
    top = res.iloc[0]
    assert top["kmer"] == motif_seq and top["rank"] == 1
    assert top["q"] < 0.05
    assert top["obs_clusters"] >= 30
    assert list(res.columns) == motif._RESULT_COLUMNS

Notes:
- Requires pandas (for the DataFrame) in addition to numpy. The shuffle uses rejection sampling of last-exit edges, which is exact Altschul-Erickson.
- `p_pooled` bins k-mers by `round(null_mean)` and pools all their null values. This assumes the k-mers in a bin are exchangeable. Ranking and BH q use `p_pooled`, over eligible k-mers within each (k, region); `p_z` is NaN when the null sd is 0.
- `run_discovery` returns only eligible k-mers (obs ≥ `min_clusters`). The region `None` is labelled "all", and `rank` is computed within each (k, region). I added optional `n_jobs` and `min_clusters` parameters.
- Shuffles are shared across k within a region (one shuffle pass per replicate).
- U is read as T. Progress goes through `logging` to stderr, via a handler attached lazily to the module logger.
- The planted-motif test uses 100 clusters of 2 sequences (300 nt), with the motif in 30 clusters, to stay under the 60 s limit; the specified 30×100×500 would exceed it with the pure-Python shuffle.
- No CLI, `--seed` or environment logging was requested, so none is included.
